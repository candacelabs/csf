package widgetfixture

import "context"

// INotifier delivers one rendered message to a downstream channel.
type INotifier interface {
	Notify(ctx context.Context, message string) error
}

// IStore keeps widget state between renders.
type IStore interface {
	Load(ctx context.Context, key string) ([]byte, error)
	Save(ctx context.Context, key string, payload []byte) error
}

// iDispatcher stays unexported and takes the lower-case prefix.
type iDispatcher interface {
	Dispatch(ctx context.Context, target string) error
}
